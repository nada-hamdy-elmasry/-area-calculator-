# -area-calculator-
simple python script to calculate area and cost
str_length=input("enter length")
str_width=input("enter width")
str_4meter=input("how much for meter?")
length=float(str_length)
width=float(str_width)
for1meter=float(str_4meter)
area=length*width
s_area=str(area)
print("the total erea is"+s_area)
perimeter=(length+width)*2
give=perimeter*for1meter

print("give the guy"+"£"+s_give)
